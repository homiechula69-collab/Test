    <script>
        // DATA
        const HERO_IMAGES = [
            "https://images.unsplash.com/photo-1515886657613-9f3515b0c78f?w=1400&q=80",
            "https://images.unsplash.com/photo-1529139574466-a303027c1d8b?w=1400&q=80",
            "https://images.unsplash.com/photo-1509631179647-0177331693ae?w=1400&q=80"
        ];

        const PRODUCTS = [
            { id: 1, name: "First Pick", price: 329, qty: 1, tag: "Solo", emoji: "🌸", img: "https://images.unsplash.com/photo-1517466787929-bc90951d0974?w=800&q=80" },
            { id: 2, name: "Brownie-Pinky Duo", price: 649, qty: 2, tag: "Duo", emoji: "🤎🩷", img: "https://images.unsplash.com/photo-1515886657613-9f3515b0c78f?w=800&q=80" },
            { id: 3, name: "Squad Set", price: 1269, qty: 4, tag: "Squad", emoji: "👯", img: "https://images.unsplash.com/photo-1529139574466-a303027c1d8b?w=800&q=80" }
        ];

        let cart = [];
        let currentSlide = 0;

        // RENDER HERO
        const heroContainer = document.getElementById('hero-slider');
        HERO_IMAGES.forEach((img, i) => {
            const slide = document.createElement('div');
            slide.className = `hero-slide ${i === 0 ? 'active' : ''}`;
            slide.style.backgroundImage = `url(${img})`;
            slide.innerHTML = `<div class="hero-overlay"></div><div class="hero-text"><h1>Custom Jersey<br>Pre-Order 2026</h1><p>ออกแบบเสื้อของคุณเอง · Homiechula</p></div>`;
            heroContainer.appendChild(slide);
        });

        setInterval(() => {
            const slides = document.querySelectorAll('.hero-slide');
            slides[currentSlide].classList.remove('active');
            currentSlide = (currentSlide + 1) % slides.length;
            slides[currentSlide].classList.add('active');
        }, 4000);

        // RENDER PRODUCTS
        const grid = document.getElementById('products-grid');
        PRODUCTS.forEach(p => {
            const card = document.createElement('div');
            card.className = 'product-card';
            card.onclick = () => openProduct(p);
            card.innerHTML = `
                <div class="product-img" style="background-image:url(${p.img})"></div>
                <div class="product-body">
                    <div class="product-tag">${p.tag}</div>
                    <div class="product-name">${p.emoji} ${p.name}</div>
                    <div class="product-price">฿${p.price}</div>
                    <button class="btn-order">สั่งเลย →</button>
                </div>
            `;
            grid.appendChild(card);
        });

        // MODAL LOGIC
        function openProduct(p) {
            const modal = document.getElementById('modal-overlay');
            const area = document.getElementById('modal-render-area');
            modal.style.display = 'flex';

            let formsHTML = '';
            for(let i=1; i<=p.qty; i++) {
                formsHTML += `
                    <div style="background:#fff; padding:1rem; border-radius:10px; margin-bottom:1rem; border:1px solid var(--border)">
                        <p style="font-weight:700; color:var(--pink-deep); margin-bottom:0.5rem;">ตัวที่ ${i}</p>
                        <div class="form-group">
                            <label class="form-label">ชื่อบนเสื้อ</label>
                            <input type="text" class="form-input" placeholder="ภาษาอังกฤษ">
                        </div>
                        <div class="form-group">
                            <label class="form-label">เบอร์เสื้อ</label>
                            <input type="number" class="form-input" placeholder="0-99">
                        </div>
                    </div>
                `;
            }

            area.innerHTML = `
                <div class="modal-body">
                    <h2 style="font-family:'Playfair Display',serif; margin-bottom:1rem;">${p.name}</h2>
                    ${formsHTML}
                    <button class="submit-btn" onclick="addToCart(${p.id})" style="width:100%; padding:1rem; background:var(--pink-deep); color:#fff; border:none; border-radius:50px; font-weight:700; cursor:pointer;">
                        เพิ่มลงตะกร้า 🛒
                    </button>
                </div>
            `;
        }

        function closeModal(e) {
            if(e.target.id === 'modal-overlay') e.target.style.display = 'none';
        }

        // CART LOGIC
        function toggleCart(show) {
            document.getElementById('cart-panel').classList.toggle('open', show);
        }

        function addToCart(id) {
            const product = PRODUCTS.find(p => p.id === id);
            cart.push(product);
            updateCart();
            document.getElementById('modal-overlay').style.display = 'none';
            showToast("เพิ่มลงตะกร้าเรียบร้อย!");
        }

        function updateCart() {
            const list = document.getElementById('cart-items');
            const badge = document.getElementById('cart-badge');
            const total = document.getElementById('cart-total');
            
            list.innerHTML = cart.length === 0 ? '<p style="text-align:center; color:var(--text-light)">ตะกร้าว่างเปล่า</p>' : '';
            
            let sum = 0;
            cart.forEach((item, index) => {
                sum += item.price;
                const div = document.createElement('div');
                div.style.cssText = "display:flex; justify-content:space-between; margin-bottom:1rem; border-bottom:1px solid #eee; padding-bottom:0.5rem;";
                div.innerHTML = `
                    <div>
                        <p style="font-weight:600; font-size:0.9rem;">${item.name}</p>
                        <p style="color:var(--pink-deep); font-size:0.8rem;">฿${item.price}</p>
                    </div>
                    <button onclick="removeItem(${index})" style="background:none; border:none; cursor:pointer;">🗑</button>
                `;
                list.appendChild(div);
            });

            badge.innerText = cart.length;
            badge.style.display = cart.length > 0 ? 'flex' : 'none';
            total.innerText = `฿${sum.toLocaleString()}`;
        }

        function removeItem(index) {
            cart.splice(index, 1);
            updateCart();
        }

        function showToast(msg) {
            const t = document.getElementById('toast');
            t.innerText = msg;
            t.style.display = 'block';
            setTimeout(() => t.style.display = 'none', 3000);
        }

        function checkout() {
            if(cart.length === 0) return alert("เลือกสินค้าก่อนนะจ๊ะ");
            showToast("ดำเนินการสั่งซื้อเรียบร้อย! ✨");
            cart = [];
            updateCart();
            toggleCart(false);
        }
    </script>
